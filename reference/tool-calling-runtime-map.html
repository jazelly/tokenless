Canonical loop / 标准闭环

visibleSchemas = project(registry, policy, context)
response = model(history, visibleSchemas)

if response is Final:
  return response

for call in response.calls:
  args = validate(call.arguments)
  result = authorize_and_execute(call.name, args)
  history += [call, result]

repeat
